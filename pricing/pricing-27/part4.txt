function priceText(p){
 if(p.free)return 'Free';
 if(p.enterprise)return 'Custom';

}

 if(p.free)return 0;

function checkoutSizeLines(p,s){
 const storage=s.tracks==='Unlimited'
   ? `${s.storage} storage - Unlimited tracks`
   : `${s.storage} storage - ${s.tracks}`;
 const members=`${s.members} members included`;
 let hubs='';
 if(p.id==='talent') hubs=`${s.slots} tracks published on Bridge Talent`;
 else if(p.id==='music') hubs=`${s.slots} tracks published on Bridge Hubs`;
 else if(p.id==='supervisor') hubs=`Bridge Talent, Sync, Library & Score included`;
 return `<div class="checkout-size-lines"><div class="checkout-size-line">${optIcon('cloud')}<span>${storage}</span></div><div class="checkout-size-line">${optIcon('team')}<span>${members}</span></div><div class="checkout-size-line">${optIcon('eye')}<span>${hubs}</span></div></div>`;
}
function renderCheckoutStack(p){
 const val=selectedPrice();
 const price=p.enterprise?'Custom':p.free?'Free':fmt(val);
 const suffix=p.sizes?'/mo':'';
 const yearlyLine=p.sizes
   ? (state.yearly?`${fmt(val*12)}/year billed yearly`:`${fmt(val*12)}/year billed monthly`)
   : (p.enterprise?'Talk to sales':'');
 const sizes=p.sizes ? `
   <div class="checkout-size-tabs" role="tablist" aria-label="${p.badge} sizes">
     ${p.sizes.map((x,i)=>`<button type="button" class="checkout-size-tab" role="tab" aria-selected="${i===state.size}" tabindex="${i===state.size?'0':'-1'}" data-size="${i}">${x.key}</button>`).join('')}
   </div>
   ${checkoutSizeLines(p,p.sizes[state.size])}` : '';
 return `<div class="checkout-stack">
   ${sizes}
   <div class="checkout-bottom"><div class="checkout-price"><div class="checkout-price-main"><span class="checkout-price-value" aria-live="polite">${price}</span>

}

  const select=()=>{
   state.plan=i; state.size=0; render(); positionCheckout();
  };
  el.addEventListener('click',select);
  el.addEventListener('keydown',e=>{
   if(e.key==='Enter'||e.key===' '){e.preventDefault();select();return;}
   if(e.key==='ArrowDown'||e.key==='ArrowRight'){e.preventDefault();focusPlan((i+1)%plans.length);}
   if(e.key==='ArrowUp'||e.key==='ArrowLeft'){e.preventDefault();focusPlan((i-1+plans.length)%plans.length);}
  });
 });
}
function focusPlan(i){
 const items=[...document.querySelectorAll('.pricing-plan')];
 items.forEach((x,j)=>x.tabIndex=j===i?0:-1);
 items[i].focus();
}
function bindSizeKeys(){
 const tabs=[...document.querySelectorAll('.checkout-size-tab')];
 tabs.forEach((tab,i)=>{
  const select=()=>{state.size=i;renderSummary();renderPlans();};
  tab.addEventListener('click',select);
  tab.addEventListener('keydown',e=>{
   let next=null;
   if(e.key==='ArrowRight'||e.key==='ArrowDown') next=(i+1)%tabs.length;
   if(e.key==='ArrowLeft'||e.key==='ArrowUp') next=(i-1+tabs.length)%tabs.length;
   if(e.key==='Home') next=0;
   if(e.key==='End') next=tabs.length-1;
   if(next!==null){
    e.preventDefault();
    state.size=next;
    renderSummary();
    renderPlans();
    requestAnimationFrame(()=>document.querySelectorAll('.checkout-size-tab')[next]?.focus());
   }
  });
 });
}

function positionCheckout(){
 const summary=document.getElementById('summary');
 if(!summary)return;
 const mobile=window.matchMedia('(max-width:900px)').matches;
 requestAnimationFrame(()=>{
  if(mobile){
   const y=summary.getBoundingClientRect().top+window.scrollY-12;
   window.scrollTo({top:Math.max(0,y),behavior:'smooth'});
  }else{
   const rect=summary.getBoundingClientRect();
   if(rect.top<24 || rect.bottom>window.innerHeight-24){
    const y=summary.getBoundingClientRect().top+window.scrollY-24;
    window.scrollTo({top:Math.max(0,y),behavior:'smooth'});
   }
   const body=summary.querySelector('.selected-plan-body');
   if(body) body.scrollTop=0;
  }
 });
}
